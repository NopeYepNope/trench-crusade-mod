event = {
	id = 3000
	random = no

	trigger = {
		technology = { type = naval_doctrine value = 4 }
		NOT = { has_country_flag = naval_doctrine_fleet_in_being }
		NOT = { has_country_flag = naval_doctrine_indirect }
		NOT = { has_country_flag = naval_doctrine_mystical }
		NOT = { has_country_flag = naval_doctrine_chosen }
		NOT = { tag = YUG }
	}

	name = "The Shape of Your Fleet"
	desc = "Your submarines have proven the potential of naval warfare beyond the surface. Your admirals debate the fundamental question: what is your fleet for?\n\nThree doctrines have emerged from the debate."

	picture = "gfx/interface/evt_naval_doctrine_choice.bmp"

	option = {
		name = "We build to be feared. Capital ships so powerful the enemy dares not engage."
		set_country_flag = naval_doctrine_fleet_in_being
		set_country_flag = naval_doctrine_chosen
		trigger_event = { id = 3001 days = 3 }
	}

	option = {
		name = "We build to be invisible. Submarines, raiders, patience."
		set_country_flag = naval_doctrine_indirect
		set_country_flag = naval_doctrine_chosen
		trigger_event = { id = 3002 days = 3 }
	}

	option = {
		name = "We build relationships. The sea itself will fight for us."
		trigger = {
			OR = {
				tag = TUR
				tag = GER
				tag = SOV
				tag = AFG
				tag = PER
				tag = IRQ
				tag = SAU
			}
		}
		set_country_flag = naval_doctrine_mystical
		set_country_flag = naval_doctrine_chosen
		trigger_event = { id = 3003 days = 3 }
	}
}
