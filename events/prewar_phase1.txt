	}

	name = "Movement in the Dark"
	desc = "The front has been quiet for eleven months. The intelligence reports that filter back from the corrupted zones speak of unusual activity — not an offensive, not yet, but something is changing in the Infernal disposition. Demon units are repositioning. Samael's death-fields are expanding. The Heretic Legion has been rotating fresh formations to the front line.\n\nEleven months of quiet. Now this.\n\nYour commanders are divided. Some believe it is preparation for a major offensive. 	picture = "gfx/interface/evt_prewar_tension.bmp"

		set_country_flag = prewar_begun

		command = { type = recon value = 0.05 }

	}
